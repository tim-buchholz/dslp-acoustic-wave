        ylabel={relative $X_h$ error},
        xmode=log,
        ymode=log,
        ymin=2e-5,
        ymax=2e0,
        xmin=1e0,
        xmax=5e2,
        unbounded coords=jump,
        legend style={at={(0.98,0.98)},anchor=north east,font=\small},
        legend columns=1,
        legend cell align={left},
        every axis plot/.append style={line width=1.3pt,mark size=2.5pt},
      ]

      \addplot+[mark=diamond*,mark options={fill=white},dstlpgreen]
        table[x=dstlp_time_loop_s,y=dstlp_rel_xh,col sep=comma] {\datapath/data/#1};
      \addlegendentry{DSLP}

      \addplot+[mark=x,mark size=3.2pt,mark options={solid,dstlpred},dstlpred]
        table[x=cn_time_loop_s,y=cn_rel_xh,col sep=comma] {\datapath/data/#1};
