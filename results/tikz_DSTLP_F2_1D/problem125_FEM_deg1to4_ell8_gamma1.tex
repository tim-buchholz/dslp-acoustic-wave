\newcommand{\DSTLPFtwoLeapfrogPlotWithLegend}[1]{%
  \addplot+[mark=triangle*,mark options={fill=white,rotate=180},dstlpblack]
    coordinates {#1}; 
  \addlegendentry{lf}  
}

\newcommand{\DSTLPFtwoLeapfrogPlot}[1]{%
  \addplot+[forget plot,mark=triangle*,mark options={fill=white,rotate=180},dstlpblack]
    coordinates {#1};
}

\newcommand{\DSTLPFtwoReferencePlotWithLegend}[1]{%
  \addplot+[mark=none,dashed,dstlpgray]
    table[x=tau,y index=5,col sep=comma] {\datapath/data/#1};
  \addlegendentry{$\mathcal{O}(\tau^2)$}
}

\newcommand{\DSTLPFtwoReferencePlot}[1]{%

}

\newcommand{\DSTLPFtwoPanelPlotsWithLegend}[1]{%

  \addlegendentry{$\text{DSLP}_{8}$}

  \addplot+[forget plot,mark=square*,mark options={solid, fill=dstlpgreen,draw=dstlpgreen},mark size=2,densely dashed,dstlpgreen]
    table[x=tau,y index=4,col sep=comma] {\datapath/data/#1};

  \addplot+[mark=x,dstlpred]
    table[x=tau,y index=1,col sep=comma] {\datapath/data/#1};
  \addlegendentry{CN}
}
